class CustomDataset(Dataset):
    def __init__(self, df, tokenizer, MAX_LEN):
        self.input_ids = []
        self.attention_masks = []

        for example in df.values:
            encoded_dict = tokenizer.encode_plus(
                example[0],                      # Input text
                example[1],                      # Target text
                max_length=MAX_LEN,
                padding='max_length',
                truncation=True,
                return_tensors='pt'
            )
            self.input_ids.append(encoded_dict['input_ids'])
            self.attention_masks.append(encoded_dict['attention_mask'])

    def __len__(self):
        return len(self.input_ids)

    def __getitem__(self, idx):
        return {
            'input_ids': self.input_ids[idx].squeeze(),
            'attention_mask': self.attention_masks[idx].squeeze()
        }


import pandas as pd
from sklearn.model_selection import train_test_split

from transformers import BartConfig, BartForConditionalGeneration, BartTokenizerFast
from transformers import DataCollatorForSeq2Seq, Trainer, TrainingArguments

import math


def train_and_save_bart_model(hyperparameters_dict, data_path="./data/dataset.csv", tokenizer_path="./data/barttokenizer/", save_to="./saved_model/"):
    TRAIN_BATCH_SIZE = hyperparameters_dict["TRAIN_BATCH_SIZE"]
    VALID_BATCH_SIZE = hyperparameters_dict["VALID_BATCH_SIZE"]
    TRAIN_EPOCHS = hyperparameters_dict["TRAIN_EPOCHS"]
    LEARNING_RATE = hyperparameters_dict["LEARNING_RATE"]
    WEIGHT_DECAY = hyperparameters_dict["WEIGHT_DECAY"]
    MAX_LEN = hyperparameters_dict["MAX_LEN"]

    config = BartConfig(
        vocab_size=hyperparameters_dict["VOCAB_SIZE"],
        max_position_embeddings=hyperparameters_dict["MAX_POSITION_EMBEDDINGS"],
        num_attention_heads=hyperparameters_dict["NUM_ATTENTION_HEADS"],
        encoder_layers=hyperparameters_dict["ENCODER_LAYERS"],
        decoder_layers=hyperparameters_dict["DECODER_LAYERS"],
        d_model=hyperparameters_dict["HIDDEN_SIZE"],
        type_vocab_size=hyperparameters_dict["TYPE_VOCAB_SIZE"]
    )

    def _model_init():
        return BartForConditionalGeneration(config=config)

    df = pd.read_csv(data_path, header=None)

    tokenizer = BartTokenizerFast.from_pretrained(tokenizer_path)

    train_df, eval_df = train_test_split(df, test_size=0.2, random_state=42)
    train_dataset = CustomDataset(train_df, tokenizer, MAX_LEN)
    eval_dataset = CustomDataset(eval_df, tokenizer, MAX_LEN)

    data_collator = DataCollatorForSeq2Seq(tokenizer=tokenizer, model=_model_init(), padding=True)

    training_args = TrainingArguments(
        output_dir=save_to,
        overwrite_output_dir=True,
        evaluation_strategy="epoch",
        save_strategy="epoch",
        num_train_epochs=TRAIN_EPOCHS,
        learning_rate=LEARNING_RATE,
        weight_decay=WEIGHT_DECAY,
        per_device_train_batch_size=TRAIN_BATCH_SIZE,
        per_device_eval_batch_size=VALID_BATCH_SIZE,
        save_total_limit=1,
        disable_tqdm=True,
        # fp16=True
    )

    trainer = Trainer(
        model_init=_model_init,
        args=training_args,
        data_collator=data_collator,
        train_dataset=train_dataset,
        eval_dataset=eval_dataset,
    )

    print("build trainer with on device:", training_args.device, "with n gpus:", training_args.n_gpu)
    trainer.train()
    print("training finished.")

    eval_results = trainer.evaluate()
    print(f">>> Perplexity: {math.exp(eval_results['eval_loss']):.2f}")

    trainer.save_model(save_to)
